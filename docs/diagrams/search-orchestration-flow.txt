2. ELASTICSEARCH QUERY STRUCTURE
=================================

{
  "query": {
    "bool": {
      "must": [
        {
          "multi_match": {
            "query": "pizza",
            "fields": ["name^3", "description^2", "cuisine"]
          }
        }
      ],
      "filter": [
        { "term": { "availability": true } },
        { "range": { "rating": { "gte": 3.5 } } },
        { "geo_distance": {
            "distance": "5km",
            "location": { "lat": 12.97, "lon": 77.59 }
        }}
      ]
    }
  },
  "aggs": {
    "cuisines": { "terms": { "field": "cuisine" } },
    "price_ranges": { "histogram": { "field": "priceRange", "interval": 1 } },
    "avg_rating": { "avg": { "field": "rating" } }
  },
  "sort": [
    { "_score": "desc" },
    { "rating": "desc" }
  ],
  "size": 20
}


3. RANKING ALGORITHM
=====================

score = (relevance * 0.30)     // Elasticsearch BM25 normalized
      + (rating    * 0.25)     // restaurant_rating / 5.0
      + (distance  * 0.20)     // 1 - (dist_km / max_radius_km)
      + (available * 0.15)     // 1.0 if open, 0.0 if closed
      + (price_fit * 0.10)     // 1.0 if in user's range, 0.5 otherwise
